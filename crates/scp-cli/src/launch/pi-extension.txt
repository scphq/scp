import { spawn } from "child_process";
import { createInterface } from "readline";
var PROTOCOL = "2025-06-18";
var CHANNEL = "notifications/claude/channel";
function swarmWords(env) {
  const raw = env.SCP_PI_SWARM;
  if (!raw) return ["swarm"];
  const words = JSON.parse(raw);
  if (!Array.isArray(words) || !words.length || !words.every((word) => typeof word === "string"))
    throw new Error("SCP_PI_SWARM must be a JSON array of strings");
  return words;
}
function shimArgv(env) {
  return [...swarmWords(env), "mcp", "--harness", "pi", "--channels", ...env.SCP_NAME ? ["--as", env.SCP_NAME] : []];
}
var McpLink = class {
  child;
  pending = /* @__PURE__ */ new Map();
  next = 1;
  closed = false;
  /** resolves once the shim process is gone */
  exited;
  constructor(argv, env, onPush, onExit) {
    this.child = spawn(argv[0], argv.slice(1), { env, stdio: ["pipe", "pipe", "ignore"] });
    this.exited = new Promise((resolve) => {
      this.child.once("exit", () => resolve());
      this.child.once("error", () => resolve());
    });
    this.child.stdin.on("error", () => {
    });
    createInterface({ input: this.child.stdout }).on("line", (line) => {
      let message;
      try {
        message = JSON.parse(line);
      } catch {
        return;
      }
      if (typeof message.id === "number" && this.pending.has(message.id)) {
        const waiter = this.pending.get(message.id);
        this.pending.delete(message.id);
        const error = message.error;
        if (error) waiter.reject(new Error(error.message ?? "mcp error"));
        else waiter.resolve(message.result ?? {});
        return;
      }
      if (message.method === CHANNEL) {
        const params = message.params ?? {};
        if (typeof params.content === "string") onPush(params.content, params.meta ?? {});
      }
    });
    this.child.once("error", () => this.fail(null, onExit));
    this.child.once("exit", (code) => this.fail(code, onExit));
  }
  fail(code, onExit) {
    if (this.closed) return;
    this.closed = true;
    for (const waiter of this.pending.values()) waiter.reject(new Error("the scp shim exited"));
    this.pending.clear();
    onExit(code);
  }
  request(method, params = {}) {
    if (this.closed) return Promise.reject(new Error("the scp shim exited"));
    const id = this.next++;
    return new Promise((resolve, reject) => {
      this.pending.set(id, { resolve, reject });
      this.child.stdin.write(`${JSON.stringify({ jsonrpc: "2.0", id, method, params })}
`);
    });
  }
  notify(method, params = {}) {
    if (!this.closed) this.child.stdin.write(`${JSON.stringify({ jsonrpc: "2.0", method, params })}
`);
  }
  /** ends the shim, resolving once it exited (or after `ms`), so a shim started next never races it for the endpoint */
  close(ms = 3e3) {
    if (!this.closed) {
      this.closed = true;
      this.child.stdin.end();
      this.child.kill();
    }
    let timer;
    return Promise.race([this.exited, new Promise((resolve) => timer = setTimeout(resolve, ms))]).finally(() => clearTimeout(timer));
  }
};
function toolResult(result) {
  const content = (result.content ?? []).flatMap(
    (part) => part.type === "text" && typeof part.text === "string" ? [{ type: "text", text: part.text }] : []
  );
  if (result.isError) throw new Error(content.map((part) => part.text).join("\n") || "the relay tool failed");
  return { content, details: {} };
}
function scp(pi) {
  let link;
  const warn = (ctx, text) => {
    if (ctx.hasUI) ctx.ui.notify(text, "warning");
  };
  pi.on("session_start", async (_event, ctx) => {
    if (process.env.SCP_PI === "0") return;
    const previous = link;
    link = void 0;
    await previous?.close();
    const current = new McpLink(
      shimArgv(process.env),
      process.env,
      (content, meta) => {
        pi.sendMessage({ customType: "scp", content, display: true, details: meta }, { triggerTurn: true, deliverAs: "steer" });
      },
      (code) => {
        if (link === current) warn(ctx, `scp: the relay shim exited (${code ?? "error"}); relay tools are unavailable`);
      }
    );
    link = current;
    try {
      await current.request("initialize", {
        protocolVersion: PROTOCOL,
        capabilities: { experimental: { "claude/channel": {} } },
        clientInfo: { name: "pi", version: "1" }
      });
      const { tools } = await current.request("tools/list");
      for (const tool of tools ?? []) {
        pi.registerTool({
          name: tool.name,
          label: tool.name,
          description: tool.description ?? tool.name,
          parameters: tool.inputSchema ?? { type: "object", properties: {} },
          async execute(_id, params) {
            if (link !== current) throw new Error("this scp session ended");
            return toolResult(await current.request("tools/call", { name: tool.name, arguments: params ?? {} }));
          }
        });
      }
      current.notify("notifications/initialized");
      const whoami = await current.request("tools/call", { name: "relay_whoami", arguments: {} });
      if (whoami.isError) warn(ctx, `scp: not on the relay: ${(whoami.content ?? []).map((part) => part.text ?? "").join(" ")}`);
    } catch (error) {
      warn(ctx, `scp: could not start the relay shim: ${error instanceof Error ? error.message : String(error)}`);
    }
  });
  pi.on("session_shutdown", async () => {
    const current = link;
    link = void 0;
    await current?.close();
  });
}
export {
  McpLink,
  scp as default,
  shimArgv,
  swarmWords,
  toolResult
};
